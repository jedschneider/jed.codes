{{- /* Build-time image pipeline: resize + WebP-compress local images. Sources live full-size in assets/; shipped derivatives are downscaled to a max width and re-encoded as WebP. Remote and unresolved images pass through untouched. */ -}} {{- $dest := .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- $maxWidth := 1600 -}} {{- if or (hasPrefix $dest "http") (hasPrefix $dest "//") (hasPrefix $dest "data:") -}} {{- else -}} {{- $img := resources.Get (strings.TrimPrefix "/" $dest) -}} {{- if $img -}} {{- $w := cond (gt $img.Width $maxWidth) $maxWidth $img.Width -}} {{- $img = $img.Resize (printf "%dx webp q80" $w) -}} {{- else -}} {{- end -}} {{- end -}}
